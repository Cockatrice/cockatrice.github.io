44

154 enum BoardType : quint32

156 MainBoard = 0,
157 SideBoard = 1,
158 Commander = 2,

165 DeckList deckList;
166
167 QString deckName = obj.value("name").toString();
168
169 QString outputText;
170 QString mainboardText;
171 QString sideboardText;
172 QTextStream mainboardStream(&mainboardText);
173 QTextStream sideboardStream(&sideboardText);
174 QTextStream outStream(&outputText);
175
176 auto commanderList = QList<QJsonObject>();
177 auto maybeboardList = QList<QJsonObject>();
178 for (auto entry : obj.value("cards").toArray()) {
179 auto card = entry.toObject();
180
181 auto collectorNumber = card.value("collectorNumber").toString();
182 auto cardName = card.value("name").toString();
183 auto boardType = static_cast<BoardType>(card.value("boardCategory").toInt());
184 auto setName = card.value("setId").toString();
185 auto quantity = card.value("quantity").toInt();
186 auto providerId = card.value("scryfallId").toString();
187
188 switch (boardType) {
189 case MainBoard:
190 mainboardStream << quantity << ' ' << cardName << " (" << setName << ") " << collectorNumber
191 << '\n';
192 break;
193 case SideBoard:
194 sideboardStream << "SB: " << quantity << ' ' << cardName << " (" << setName << ") "
195 << collectorNumber << '\n';
196 break;
197 case MaybeBoard:
198 maybeboardList.append(card);
199 break;
200 case Commander:
201 commanderList.append(card);
202 break;
203 default:
204 qCWarning(InterfaceJsonDeckParserLog)
205 << "Unknown ManaBox Zone" << card.value("boardCategory").toInt() << "for" << cardName;
206 break;
207 }
208 }
209
210 outStream << mainboardText << sideboardText;
211
212 deckList.loadFromStream_Plain(outStream, false, CardNameNormalizer());
213 deckList.setName(deckName);
214 deckList.forEachCard(CardNodeFunction::ResolveProviderId());
215
216 for (const auto &maybeboardCard : maybeboardList) {
217 QString cardName = maybeboardCard.value("name").toString();
218 QString setName = maybeboardCard.value("setId").toString().toUpper();
219 QString collectorNumber = maybeboardCard.value("collectorNumber").toString();
220 QString providerId = maybeboardCard.value("scryfallId").toString();
221
222 for (int x = 0; x < maybeboardCard.value("quantity").toInt(); x++) {
223 deckList.addCard(cardName, DECK_ZONE_MAYBEBOARD, -1, setName, collectorNumber, providerId);
224 }
225 }
226 for (const auto &commanderCard : commanderList) {
227
228 QString commanderName = commanderCard.value("name").toString();
229 QString setName = commanderCard.value("setId").toString().toUpper();
230 QString collectorNumber = commanderCard.value("collectorNumber").toString();
231 QString providerId = commanderCard.value("scryfallId").toString();
232
233 deckList.setBannerCard({commanderName, providerId});
234 deckList.addCard(commanderName, DECK_ZONE_MAIN, -1, setName, collectorNumber, providerId);
235 }
