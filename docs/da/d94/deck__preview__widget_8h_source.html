123protected:
124 bool eventFilter(QObject *obj, QEvent *event) override
125 {
126 if (event->type() == QEvent::Wheel) {
127 if (auto *combo = qobject_cast<QComboBox *>(obj)) {
128 // If popup is not open, forward event to parent scroll area
129 if (!combo->view()->isVisible()) {
130 // Try to find a scrollable parent and manually send the event
131 QWidget *parent = combo->parentWidget();
132 while (parent) {
133 if (auto *scroll = qobject_cast<QAbstractScrollArea *>(parent)) {
134 QApplication::sendEvent(scroll->viewport(), event);
135 return true; // Mark event as handled
136 }
137 parent = parent->parentWidget();
138 }
139 // If no scrollable parent found, just block
140 return true;
141 }
142 }
143 }
144 return QObject::eventFilter(obj, event);
145 }
